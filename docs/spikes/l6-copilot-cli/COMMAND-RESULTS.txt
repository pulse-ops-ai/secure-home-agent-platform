Exact harness invocation shape (hosted gpt-5.4; credential supplied by OS store, never in argv):
copilot -C <scratch-workspace> -p <prompt> --model gpt-5.4 --output-format json --stream off \
  --no-color --no-custom-instructions --no-auto-update --disable-builtin-mcps \
  --no-remote --no-remote-export --no-ask-user --session-id <uuid> \
  --share=<run>/share.md --log-dir <run>/logs [case-specific --available-tools/--allow-tool/--deny-tool/--secret-env-vars]
env: COPILOT_HOME=<scratch-home> COPILOT_OTEL_FILE_EXPORTER_PATH=<run>/otel.jsonl COPILOT_OTEL_EXPORTER_TYPE=file OTEL_INSTRUMENTATION_GENAI_CAPTURE_MESSAGE_CONTENT=<true|false>
